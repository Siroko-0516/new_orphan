cmake_minimum_required(VERSION 3.22.1)
project(giljabi LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=4 link_pool=2)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
set(LLAMA_SRC "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp-omni")
if(NOT EXISTS "${LLAMA_SRC}/CMakeLists.txt")
  message(FATAL_ERROR "Run python scripts/prepare_engine.py from the android directory first.")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
option(GILJABI_VISION_VULKAN "Build optional Vulkan vision acceleration" ON)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a" AND GILJABI_VISION_VULKAN)
  set(GILJABI_VULKAN_HEADERS "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/vulkan-headers")
  if(NOT EXISTS "${GILJABI_VULKAN_HEADERS}/include/vulkan/vulkan.hpp" OR
     NOT EXISTS "${GILJABI_VULKAN_HEADERS}/cmake/SPIRV-HeadersConfig.cmake")
    message(FATAL_ERROR "Run python android/scripts/prepare_vulkan_headers.py before the native build, or set GILJABI_VISION_VULKAN=OFF.")
  endif()
  # The shader compiler runs on the build host, never on the Android device.
  set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/${ANDROID_HOST_TAG}/glslc" CACHE FILEPATH "NDK host shader compiler" FORCE)
  if(NOT EXISTS "${Vulkan_GLSLC_EXECUTABLE}")
    if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
      set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/linux-x86_64/glslc" CACHE FILEPATH "NDK host shader compiler" FORCE)
    elseif(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
      set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/darwin-x86_64/glslc" CACHE FILEPATH "NDK host shader compiler" FORCE)
    elseif(CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows")
      set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/windows-x86_64/glslc.exe" CACHE FILEPATH "NDK host shader compiler" FORCE)
    endif()
  endif()
  if(NOT EXISTS "${Vulkan_GLSLC_EXECUTABLE}")
    message(FATAL_ERROR "The selected NDK does not include a host glslc compiler.")
  endif()
  # Vulkan 1.1 loader symbols are public from Android API 28. Only this
  # optional module links that stub; app/JNI/CPU remain API 26 compatible.
  # API 26-27 cannot load the optional module and retain the CPU backend.
  set(Vulkan_LIBRARY "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-android/28/libvulkan.so" CACHE FILEPATH "Optional Vulkan loader stub" FORCE)
  set(Vulkan_INCLUDE_DIR "${GILJABI_VULKAN_HEADERS}/include" CACHE PATH "Pinned Vulkan C/C++ headers" FORCE)
  set(SPIRV-Headers_DIR "${GILJABI_VULKAN_HEADERS}/cmake" CACHE PATH "Pinned SPIR-V headers" FORCE)
  include_directories(BEFORE "${GILJABI_VULKAN_HEADERS}/include")
  set(GGML_VULKAN ON CACHE BOOL "" FORCE)
endif()
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  set(GGML_CPU_ARM_ARCH "" CACHE STRING "" FORCE)
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SRC}" "${CMAKE_BINARY_DIR}/llama")
add_library(giljabi_native SHARED llama_jni.cpp)
target_include_directories(giljabi_native PRIVATE "${LLAMA_SRC}/common" "${LLAMA_SRC}/include" "${LLAMA_SRC}/ggml/include" "${LLAMA_SRC}/ggml/src" "${LLAMA_SRC}/tools/mtmd" "${LLAMA_SRC}/vendor")
target_link_libraries(giljabi_native llama llama-common mtmd android log)
